\begin{operator}[Golden scaling, $\phigr$-SCL]\label{op:phi}
For $n\in\mathbb{Z}$ define the golden scaling operator
\begin{equation}
  S_{\phigr}^{\,n} : \mathbb{R}\to\mathbb{R},\qquad
  S_{\phigr}^{\,n}(x) = \phigr^{\,n}\,x,\qquad
  \phigr = \frac{1+\sqrt5}{2}.
\end{equation}
Its byte-quantized signature is $\sigma(n) = \big\lfloor \phigr^{\,n}\big\rfloor
\bmod 256 \in \Zmod{256}$, computed exactly via Lucas numbers
$L_n$ since $\lfloor\phigr^{\,n}\rfloor = L_n$ for odd $n\ge 1$.
In particular $\sigma(101) = L_{101}\bmod 256 = 75$.
\end{operator}

\begin{operator}[Differentiation stencil, $\DIAG(1,-2,1)$]\label{op:diag}
Let $\mathbf{d}=(1,-2,1)$. For $\bm{v}\in\mathbb{R}^n$ ($n\ge3$) define
\begin{equation}
  (\DIAG\,\bm{v})_i \;=\; v_{i-1} - 2v_i + v_{i+1},
  \qquad i = 2,\dots,n-1 .
\end{equation}
The stencil is a null vector, $\textstyle\sum_k d_k = 1-2+1 = 0$, so
constant and affine signals map to $\bm{0}$:
\begin{equation}
  \DIAG(a,\,b,\,a) = 2(a-b) = 0 \iff a=b .
\end{equation}
$\DIAG$ is the discrete second derivative and an edge/contrast detector.
\end{operator}

\begin{operator}[Triality automorphism, $\TRI$]\label{op:tri}
Let $\mathrm{Out}(\mathrm{Spin}(8))\cong S_3$ act on the three
$8$-dimensional representations $(\mathbf{8}_v,\mathbf{8}_s,\mathbf{8}_c)$
(COSMO-D-004).
The triality operator $T$ is the order-3 outer automorphism
\begin{equation}
  T:\ \mathbf{8}_v \mapsto \mathbf{8}_s \mapsto \mathbf{8}_c \mapsto \mathbf{8}_v,
  \qquad T^3 = \mathrm{id}.
\end{equation}
COSMO symbolically associates this three-way representation symmetry with
capsid branching (COSMO-D-011). No established biological mechanism is claimed;
a future quantitative bridge would require the falsifiable protocol recorded

\begin{operator}[Ouroboros recursion, $\Ouro$]\label{op:ouro}

define the recursion $R:\Zmod{256}\to\Zmod{256}$,
